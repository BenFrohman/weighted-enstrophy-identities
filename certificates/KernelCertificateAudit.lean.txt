import Lean
import «WeightedEnstrophy»
import «WeightedEnstrophy».«Abstract»
import «WeightedEnstrophy».«Fourier».«Basic»
import «WeightedEnstrophy».«Fourier».«Strain»
import «WeightedEnstrophy».«Lyapunov»
import «WeightedEnstrophy».«Smooth»
import «WeightedEnstrophy».«TorusCalculus»
import «WeightedEnstrophy».«TrigModel»

open Lean Elab Command

/-- Prints `#print axioms`-style lines for every theorem declared in the listed modules. -/
elab "#kernel_certificate_audit" : command => do
  let env ← getEnv
  let mods : Array Name := #[`«WeightedEnstrophy», `«WeightedEnstrophy».«Abstract», `«WeightedEnstrophy».«Fourier».«Basic», `«WeightedEnstrophy».«Fourier».«Strain», `«WeightedEnstrophy».«Lyapunov», `«WeightedEnstrophy».«Smooth», `«WeightedEnstrophy».«TorusCalculus», `«WeightedEnstrophy».«TrigModel»]
  for m in mods do
    match env.getModuleIdx? m with
    | none => IO.println s!"MODULE NOT LOADED {m}"
    | some i =>
      for n in (env.header.moduleData[i.toNat]!).constNames do
        match env.find? n with
        | some (.thmInfo _) =>
          if !n.isInternal then
            let ax ← collectAxioms n
            let axs := ", ".intercalate (ax.toList.map toString)
            if ax.isEmpty then IO.println s!"'{n}' does not depend on any axioms [module {m}]"
            else IO.println s!"'{n}' depends on axioms: [{axs}] [module {m}]"
        | _ => pure ()

#kernel_certificate_audit
